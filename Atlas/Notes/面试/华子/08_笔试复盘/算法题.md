import java.util.*;

public class Main {
    public static void main(String[] args) {
        // 读取输入
        Scanner sc = new Scanner(System.in);
        int K = sc.nextInt();          // 最大合并次数
        sc.nextLine();                 // 消耗掉数字后面的换行符
        String S = sc.nextLine().trim(); // 原始字符串

        // 初始 Token 序列：将字符串的每个字符视为一个独立的 Token
        List<String> tokens = new ArrayList<>();
        for (char c : S.toCharArray()) {
            tokens.add(String.valueOf(c));
        }

        // 进行最多 K 轮合并
        for (int step = 0; step < K; step++) {
            // 如果只剩下一个 Token，无法再合并，直接退出
            if (tokens.size() < 2) break;

            // ---------- 步骤1：统计所有相邻 Token 对的出现频率 ----------
            Map<Pair, Integer> freq = new HashMap<>();
            for (int i = 0; i < tokens.size() - 1; i++) {
                Pair pair = new Pair(tokens.get(i), tokens.get(i + 1));
                freq.put(pair, freq.getOrDefault(pair, 0) + 1);
            }
            // 如果没有相邻对（理论上 size<2 时已退出，此处防御）
            if (freq.isEmpty()) break;

            // ---------- 步骤2：找出频率最高的相邻对（平局按字典序最小） ----------
            int maxFreq = 0;
            Pair best = null;
            for (Map.Entry<Pair, Integer> entry : freq.entrySet()) {
                int f = entry.getValue();
                Pair p = entry.getKey();
                if (f > maxFreq) {
                    maxFreq = f;
                    best = p;
                } else if (f == maxFreq && best != null && p.compareTo(best) < 0) {
                    // 频率相同且当前对字典序更小，则更新
                    best = p;
                } else if (f == maxFreq && best == null) {
                    best = p;
                }
            }

            // 终止条件：最高频率为 1（即所有相邻对都只出现一次）
            if (maxFreq <= 1) break;

            // ---------- 步骤3：合并所有出现的最佳对（从左到右贪心） ----------
            List<String> newTokens = new ArrayList<>();
            for (int i = 0; i < tokens.size(); ) {
                // 检查当前位置 i 和下一个位置 i+1 是否构成最佳对
                if (i + 1 < tokens.size() &&
                    tokens.get(i).equals(best.first) &&
                    tokens.get(i + 1).equals(best.second)) {
                    // 合并这两个 Token，生成新 Token（直接拼接字符串）
                    newTokens.add(best.first + best.second);
                    i += 2; // 跳过已经合并的两个位置
                } else {
                    // 不匹配，直接保留当前 Token
                    newTokens.add(tokens.get(i));
                    i++;
                }
            }
            // 用合并后的新序列替换原序列，进入下一轮迭代
            tokens = newTokens;
        }

        // 输出最终 Token 序列，Token 之间用空格分隔
        System.out.println(String.join(" ", tokens));
    }

    /**
     * 表示一个相邻 Token 对 (first, second)
     * 实现了 Comparable 接口，用于按字典序比较两个 Pair
     * 比较规则：先比较 first，再比较 second
     */
    static class Pair implements Comparable<Pair> {
        String first;
        String second;

        Pair(String first, String second) {
            this.first = first;
            this.second = second;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Pair pair = (Pair) o;
            return Objects.equals(first, pair.first) && Objects.equals(second, pair.second);
        }

        @Override
        public int hashCode() {
            return Objects.hash(first, second);
        }

        /**
         * 字典序比较：先比较 first，如果相等再比较 second
         */
        @Override
        public int compareTo(Pair o) {
            int cmp = first.compareTo(o.first);
            if (cmp != 0) return cmp;
            return second.compareTo(o.second);
        }
    }
}
